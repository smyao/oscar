cmake_minimum_required(VERSION 3.18)
project(oscar_aligned_math LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "research build" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "Requires official ascend910b4 SDK")
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
set(SOURCES ../../kernels/attention_cv_aligned_decode.cpp ../../kernels/store_aligned_int2.cpp)
if(OSCAR_ALIGNED_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  add_executable(oscar_aligned_math_cpu probe.cpp ${SOURCES} ../../kernels/attention_tasks.cpp ../../kernels/merge_lse.cpp)
  target_link_libraries(oscar_aligned_math_cpu PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(oscar_aligned_math_cpu SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(oscar_aligned_math_cpu PRIVATE -O0 -g)
endif()
