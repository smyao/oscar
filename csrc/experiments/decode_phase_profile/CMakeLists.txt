# Archive #140/#144/#148/#154/#155/D.4: diagnostic-only exact-source clocks.
# No production target or dispatch changes. Build only after generate.py --check.
cmake_minimum_required(VERSION 3.18)
project(oscar_decode_phase_profile LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "isolated profiling build" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4")
  message(FATAL_ERROR "Requires SOC_VERSION=ascend910b4")
endif()
if(NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "ASCEND_HOME_PATH is required")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" --check
  RESULT_VARIABLE profile_check_rc)
if(NOT profile_check_rc EQUAL 0)
  message(FATAL_ERROR "Pinned source/generated profiling clone check failed")
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
foreach(candidate "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake"
                  "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
  if(EXISTS "${candidate}/ascendc.cmake")
    set(ASCENDC_CMAKE_DIR "${candidate}")
    break()
  endif()
endforeach()
if(NOT ASCENDC_CMAKE_DIR)
  message(FATAL_ERROR "CANN ascendc.cmake missing")
endif()
include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
ascendc_library(oscar_decode_phase_profile SHARED simd_profile.cpp pipeline_profile.cpp)
if(OSCAR_CAMODEL_MARKERS)
  ascendc_compile_definitions(oscar_decode_phase_profile PRIVATE OSCAR_CAMODEL_MARKERS=1)
endif()
