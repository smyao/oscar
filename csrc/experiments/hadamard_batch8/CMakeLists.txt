cmake_minimum_required(VERSION 3.18)
project(oscar_hadamard_batch8 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "isolated experiment" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "Requires ascend910b4 and ASCEND_HOME_PATH")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" --check
  --manifest "${CMAKE_CURRENT_BINARY_DIR}/frozen_source_binding.json" RESULT_VARIABLE rc)
if(NOT rc EQUAL 0)
  message(FATAL_ERROR "Hadamard source identity check failed")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/entrypoints.py" --check RESULT_VARIABLE rc)
if(NOT rc EQUAL 0)
  message(FATAL_ERROR "Writer entrypoint identity check failed")
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
set(writer_sources ../striped_store_simd_pack/writer_baseline.cpp ../striped_store_simd_pack/writer_pack_init.cpp writer_batch.cpp)
set(rotate_sources ../../kernels/rotate_clip_store.cpp rotate_init.cpp rotate_batch.cpp)
if(OSCAR_BATCH8_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  add_executable(oscar_batch8_store_cpu store_cpu.cpp ${writer_sources})
  set(cpu_targets oscar_batch8_store_cpu)
  if(NOT OSCAR_BATCH8_STORE_ONLY)
    add_executable(oscar_batch8_rotate_cpu rotate_cpu.cpp ${rotate_sources})
    list(APPEND cpu_targets oscar_batch8_rotate_cpu)
  endif()
  foreach(target ${cpu_targets})
    target_link_libraries(${target} PRIVATE tikicpulib::Ascend910B4)
    target_include_directories(${target} SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
    target_compile_options(${target} PRIVATE -O0 -g)
  endforeach()
else()
  foreach(path "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake" "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
    if(EXISTS "${path}/ascendc.cmake")
      set(ASCENDC_CMAKE_DIR "${path}")
      break()
    endif()
  endforeach()
  if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "CANN ascendc.cmake missing")
  endif()
  include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
  ascendc_library(oscar_hadamard_batch8 SHARED ${writer_sources} ${rotate_sources})
endif()
