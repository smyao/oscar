# Archive G7/G8/#79-85/#126/#145/#154, D.4: independent writer A/B/C only.
cmake_minimum_required(VERSION 3.18)
project(oscar_striped_store_simd_pack LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "isolated diagnostic" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "Requires SOC_VERSION=ascend910b4 and ASCEND_HOME_PATH")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py"
  --check --frozen-only --manifest "${CMAKE_CURRENT_BINARY_DIR}/frozen_source_binding.json" RESULT_VARIABLE rc)
if(NOT rc EQUAL 0)
  message(FATAL_ERROR "Writer clone identity check failed")
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
if(OSCAR_SIMD_PACK_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  add_executable(oscar_simd_pack_store_cpu store_cpu.cpp writer_baseline.cpp writer_pack.cpp writer_pack_init.cpp)
  add_executable(oscar_simd_pack_exhaustive_cpu pack_cpu.cpp pack_kernel.cpp)
  foreach(target oscar_simd_pack_store_cpu oscar_simd_pack_exhaustive_cpu)
    target_link_libraries(${target} PRIVATE tikicpulib::Ascend910B4)
    target_include_directories(${target} SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
    target_compile_options(${target} PRIVATE -O0 -g)
  endforeach()
else()
  foreach(candidate "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake" "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
    if(EXISTS "${candidate}/ascendc.cmake")
      set(ASCENDC_CMAKE_DIR "${candidate}")
      break()
    endif()
  endforeach()
  if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "CANN ascendc.cmake missing")
  endif()
  include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
  ascendc_library(oscar_striped_store_simd_pack SHARED writer_baseline.cpp writer_pack.cpp writer_pack_init.cpp pack_kernel.cpp)
endif()
