cmake_minimum_required(VERSION 3.18)
project(oscar_fp32_nz_b_matmul LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "isolated NZ B build" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "Requires explicit ascend910b4 and ASCEND_HOME_PATH")
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/kernel.cpp" kernel_sha)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../../kernels/oscar_common.h" common_sha)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../../kernels/attention_cv_decode_bundle.cpp" baseline_sha)
if(NOT baseline_sha STREQUAL "d8171f0403cf176087e1c76c64e454e6fb1e74fd79252aab3b59e848cca19401")
  message(FATAL_ERROR "Re-audit changed decode configuration source")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source_binding.json"
  "{\"kernel_sha256\":\"${kernel_sha}\",\"common_sha256\":\"${common_sha}\",\"baseline_sha256\":\"${baseline_sha}\",\"basic\":[64,64,128],\"single\":[128,256,256],\"C_dtype\":\"float32\",\"hf32\":false,\"B_formats\":[\"ND\",\"NZ\"],\"C0\":8}\n")
if(OSCAR_NZ_B_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  add_executable(oscar_fp32_nz_b_matmul_cpu probe.cpp kernel.cpp)
  target_link_libraries(oscar_fp32_nz_b_matmul_cpu PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(oscar_fp32_nz_b_matmul_cpu SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(oscar_fp32_nz_b_matmul_cpu PRIVATE -O0 -g)
else()
  foreach(path "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake" "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
    if(EXISTS "${path}/ascendc.cmake")
      set(ASCENDC_CMAKE_DIR "${path}")
      break()
    endif()
  endforeach()
  if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "CANN ascendc.cmake missing")
  endif()
  include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
  ascendc_library(oscar_fp32_nz_b_matmul SHARED kernel.cpp)
endif()
