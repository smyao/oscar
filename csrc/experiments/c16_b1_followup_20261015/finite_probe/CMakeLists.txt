cmake_minimum_required(VERSION 3.18)
project(oscar_c16_finite_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "Requires ascend910b4 official SDK")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../finite_probe.py" --check RESULT_VARIABLE rc)
if(NOT rc EQUAL 0)
  message(FATAL_ERROR "Exact Softmax extraction identity failed")
endif()
find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
add_executable(oscar_c16_finite_probe main.cpp baseline.cpp candidate.cpp)
target_link_libraries(oscar_c16_finite_probe PRIVATE tikicpulib::Ascend910B4)
target_include_directories(oscar_c16_finite_probe SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
target_compile_options(oscar_c16_finite_probe PRIVATE -O0 -g)
