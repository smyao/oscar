# Archive G4/G7-G13/#126/#145/#148/#154/#157: independent BF16 cluster build.
cmake_minimum_required(VERSION 3.18)
project(oscar_bf16_clusters LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "isolated BF16 build" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "Requires official ascend910b4 SDK")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../../tools/generate_bf16_clusters.py" --check RESULT_VARIABLE rc)
if(NOT rc EQUAL 0)
  message(FATAL_ERROR "BF16 cluster fixed-source identity failed")
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
if(OSCAR_BF16_CLUSTERS_CPU_DEBUG)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/harness.py" --check RESULT_VARIABLE hrc)
  if(NOT hrc EQUAL 0)
    message(FATAL_ERROR "Independent BF16 checker identity failed")
  endif()
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  add_executable(oscar_bf16_clusters_cpu probe.cpp ../../kernels/attention_cv_bf16_cluster4.cpp ../../kernels/attention_cv_bf16_cluster16.cpp ../../kernels/attention_tasks.cpp ../../kernels/merge_lse.cpp)
  target_link_libraries(oscar_bf16_clusters_cpu PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(oscar_bf16_clusters_cpu SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(oscar_bf16_clusters_cpu PRIVATE -O0 -g)
  return()
endif()
foreach(path "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake" "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
  if(EXISTS "${path}/ascendc.cmake")
    set(ASCENDC_CMAKE_DIR "${path}")
    break()
  endif()
endforeach()
if(NOT ASCENDC_CMAKE_DIR)
  message(FATAL_ERROR "CANN ascendc.cmake missing")
endif()
include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
ascendc_library(oscar_bf16_clusters SHARED ../../kernels/attention_cv_bf16_cluster4.cpp ../../kernels/attention_cv_bf16_cluster16.cpp)
