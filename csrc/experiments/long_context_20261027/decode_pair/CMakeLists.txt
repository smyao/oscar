# Archive G7-G13/#79-85/#126/#145/#164: fresh, isolated functional build.
# D.4: two complete bounded decode readers; no performance measurements.
cmake_minimum_required(VERSION 3.18)
project(oscar_long_context_decode_pair LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "functional build" FORCE)
get_filename_component(OSCAR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "Requires local official ascend910b4 SDK")
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
if(OSCAR_DECODE_PAIR_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  foreach(arm before after)
    if(arm STREQUAL "before")
      set(reader "${OSCAR_ROOT}/csrc/kernels/long_context_decode_baseline.cpp")
    else()
      set(reader "${OSCAR_ROOT}/csrc/kernels/attention_cv_official_decode.cpp")
    endif()
    add_executable(oscar_decode_${arm}
      "${OSCAR_ROOT}/csrc/experiments/official_math_20261021/probe.cpp"
      "${reader}"
      "${OSCAR_ROOT}/csrc/kernels/store_aligned_int2.cpp"
      "${OSCAR_ROOT}/csrc/kernels/attention_official_tasks.cpp"
      "${OSCAR_ROOT}/csrc/kernels/merge_lse.cpp")
    target_link_libraries(oscar_decode_${arm} PRIVATE tikicpulib::Ascend910B4)
    target_include_directories(oscar_decode_${arm} SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
    target_compile_options(oscar_decode_${arm} PRIVATE -O0 -g)
  endforeach()
  return()
endif()
foreach(path "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake" "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
  if(EXISTS "${path}/ascendc.cmake")
    set(ASCENDC_CMAKE_DIR "${path}")
    break()
  endif()
endforeach()
if(NOT ASCENDC_CMAKE_DIR)
  message(FATAL_ERROR "CANN ascendc.cmake missing")
endif()
include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
ascendc_library(oscar_long_context_decode SHARED "${OSCAR_ROOT}/csrc/kernels/attention_cv_official_decode.cpp")
