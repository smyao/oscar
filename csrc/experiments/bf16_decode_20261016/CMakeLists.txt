# Archive G7-G13/#126/#145/#154: fresh, isolated official CPU/CANN builds.
cmake_minimum_required(VERSION 3.18)
project(oscar_bf16_decode LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "research build" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "Requires official ascend910b4 SDK")
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
set(SOURCES ../../kernels/attention_cv_bf16_decode.cpp)
if(OSCAR_BF16_DECODE_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  add_executable(oscar_bf16_decode_cpu probe.cpp ${SOURCES} ../../kernels/attention_tasks.cpp ../../kernels/merge_lse.cpp)
  target_link_libraries(oscar_bf16_decode_cpu PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(oscar_bf16_decode_cpu SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(oscar_bf16_decode_cpu PRIVATE -O0 -g)
else()
  foreach(path "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake" "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
    if(EXISTS "${path}/ascendc.cmake")
      set(ASCENDC_CMAKE_DIR "${path}")
      break()
    endif()
  endforeach()
  if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "CANN ascendc.cmake missing")
  endif()
  include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
  # Same compiler/library holds the unchanged main FP32 and new BF16 arms.
  ascendc_library(oscar_bf16_decode SHARED ${SOURCES}
    ../../kernels/attention_cv_hybrid_l1_decode.cpp
    ../../kernels/attention_cv_mdl_decode.cpp)
endif()
