# Archive #13-22/#126/#145/#154 and D.4: isolated current-helper parity/cycles.
cmake_minimum_required(VERSION 3.18)
project(oscar_unpack32_helper LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "isolated build" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "Requires SOC_VERSION=ascend910b4 and ASCEND_HOME_PATH")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" --check RESULT_VARIABLE rc)
if(NOT rc EQUAL 0)
  message(FATAL_ERROR "Current helper source identity failed")
endif()
if(OSCAR_UNPACK32_CV)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_c4.py" --check RESULT_VARIABLE cv_rc)
  if(NOT cv_rc EQUAL 0)
    message(FATAL_ERROR "C4 source restoration/math/view checks failed")
  endif()
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
if(OSCAR_UNPACK32_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  if(OSCAR_UNPACK32_CV)
    set(cpu_target oscar_c4_unpack32_cpu)
    add_executable(${cpu_target} cv_probe.cpp c4_baseline.cpp c4_candidate.cpp ../../kernels/attention_tasks.cpp)
  else()
    set(cpu_target oscar_unpack32_helper_cpu)
    add_executable(${cpu_target} probe.cpp helper_kernel.cpp)
  endif()
  target_link_libraries(${cpu_target} PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(${cpu_target} SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(${cpu_target} PRIVATE -O0 -g)
else()
  foreach(path "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake" "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
    if(EXISTS "${path}/ascendc.cmake")
      set(ASCENDC_CMAKE_DIR "${path}")
      break()
    endif()
  endforeach()
  if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "CANN ascendc.cmake missing")
  endif()
  include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
  if(OSCAR_UNPACK32_CV)
    ascendc_library(oscar_c4_unpack32 SHARED c4_baseline.cpp c4_candidate.cpp)
  else()
    ascendc_library(oscar_unpack32_helper SHARED helper_kernel.cpp)
  endif()
endif()
