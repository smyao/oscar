# Archive G7-G13/#126/#140/#145/#154: isolated current-family finite proof.
cmake_minimum_required(VERSION 3.18)
project(oscar_decode_finite LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "research build" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "Requires official ascend910b4 SDK")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" --check
 --build-binding "${CMAKE_BINARY_DIR}/source_binding.json" RESULT_VARIABLE rc)
if(NOT rc EQUAL 0)
  message(FATAL_ERROR "Pinned production/source binding failed")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/harness.py" --check RESULT_VARIABLE rc)
if(NOT rc EQUAL 0)
  message(FATAL_ERROR "Complete checker binding failed")
endif()
set(SOURCES q4_baseline.cpp q4_candidate.cpp q1_baseline.cpp q1_candidate.cpp)
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
if(OSCAR_DECODE_FINITE_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  set(PROBE probe.cpp)
  if(OSCAR_DECODE_FINITE_DIAGNOSTIC)
    set(SOURCES q4_diagnostic.cpp q4_candidate.cpp q1_baseline.cpp q1_candidate.cpp)
    set(PROBE diagnostic_probe.cpp)
  endif()
  add_executable(oscar_decode_finite_cpu ${PROBE} ${SOURCES} ../../kernels/attention_tasks.cpp ../../kernels/merge_lse.cpp)
  target_compile_definitions(oscar_decode_finite_cpu PRIVATE
   OSCAR_CANDIDATE_Q4_KERNEL=oscar_decode_finite_q4_candidate_kernel
   OSCAR_CANDIDATE_Q1_KERNEL=oscar_decode_finite_q1_candidate_q1_kernel
   OSCAR_CANDIDATE_WORKSPACE_BYTES_PER_CORE=1146880 OSCAR_CANDIDATE_LABEL="finite_readback")
  target_link_libraries(oscar_decode_finite_cpu PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(oscar_decode_finite_cpu SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(oscar_decode_finite_cpu PRIVATE -O0 -g)
else()
  foreach(path "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake" "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
    if(EXISTS "${path}/ascendc.cmake")
      set(ASCENDC_CMAKE_DIR "${path}")
      break()
    endif()
  endforeach()
  if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "CANN ascendc.cmake missing")
  endif()
  include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
  ascendc_library(oscar_decode_finite SHARED ${SOURCES})
endif()
