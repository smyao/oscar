# Archive G7-G13/#134/#140/#145/#154: real extracted decode Softmax CPU proof.
cmake_minimum_required(VERSION 3.18)
project(oscar_decode_finite_methods LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
 message(FATAL_ERROR "Requires official ascend910b4 SDK")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../finite_probe.py" --check RESULT_VARIABLE rc)
if(NOT rc EQUAL 0)
 message(FATAL_ERROR "Actual method extraction binding failed")
endif()
find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
foreach(family q4 q1)
 add_executable(oscar_decode_finite_${family} main.cpp ${family}_baseline.cpp ${family}_candidate.cpp)
 if(family STREQUAL "q4")
  set(queries 4)
 else()
  set(queries 1)
 endif()
 target_compile_definitions(oscar_decode_finite_${family} PRIVATE FINITE_QUERIES=${queries}
  FINITE_BASELINE=finite_probe_${family}_baseline FINITE_CANDIDATE=finite_probe_${family}_candidate)
 target_link_libraries(oscar_decode_finite_${family} PRIVATE tikicpulib::Ascend910B4)
 target_include_directories(oscar_decode_finite_${family} SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
 target_compile_options(oscar_decode_finite_${family} PRIVATE -O0 -g)
endforeach()
