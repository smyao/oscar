# Archive #126/#132-135/#145/#148/#154/#155: isolated C4 identity/diagnostic build.
cmake_minimum_required(VERSION 3.18)
project(oscar_c4_phase_profile LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "CANN merge build" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "Requires ASCEND_HOME_PATH and SOC_VERSION=ascend910b4")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" --check RESULT_VARIABLE check_rc)
if(NOT check_rc EQUAL 0)
  message(FATAL_ERROR "Pinned C4 normal/profile identity failed")
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
if(OSCAR_C4_PREPARE_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  add_executable(oscar_c4_prepare_cpu prepare_cpu.cpp ../../kernels/attention_tasks.cpp)
  target_link_libraries(oscar_c4_prepare_cpu PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(oscar_c4_prepare_cpu SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(oscar_c4_prepare_cpu PRIVATE -O0 -g)
else()
  foreach(candidate "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake" "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
    if(EXISTS "${candidate}/ascendc.cmake")
      set(ASCENDC_CMAKE_DIR "${candidate}")
      break()
    endif()
  endforeach()
  if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "CANN ascendc.cmake missing")
  endif()
  include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
  ascendc_library(oscar_c4_phase_profile SHARED normal.cpp profile.cpp)
  if(OSCAR_CAMODEL_MARKERS)
    ascendc_compile_definitions(oscar_c4_phase_profile PRIVATE OSCAR_CAMODEL_MARKERS=1)
  endif()
endif()
