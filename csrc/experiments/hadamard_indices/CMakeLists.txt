# Archive #81-83/#125/#154: standalone CANN/CPU proof, no production linkage.
cmake_minimum_required(VERSION 3.18)
project(oscar_hadamard_indices LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "requires explicit ascend910b4 and CANN path")
endif()
set(CMAKE_BUILD_TYPE Release CACHE STRING "CANN build type" FORCE)
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "SOC" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
if(OSCAR_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  add_executable(oscar_hadamard_indices_cpu probe.cpp kernel.cpp)
  target_link_libraries(oscar_hadamard_indices_cpu PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(oscar_hadamard_indices_cpu SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(oscar_hadamard_indices_cpu PRIVATE -O0 -g)
else()
  foreach(candidate "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake" "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
    if(EXISTS "${candidate}/ascendc.cmake")
      set(ASCENDC_CMAKE_DIR "${candidate}")
      break()
    endif()
  endforeach()
  include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
  ascendc_library(oscar_hadamard_indices SHARED kernel.cpp)
endif()
