# Bounded rotated-FP32 producer experiment; no production binder or route.
# Archive #126/#129/#145/#148-155 and startup D.4.
cmake_minimum_required(VERSION 3.18)
project(oscar_shared_fp32_history LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT SOC_VERSION STREQUAL "ascend910b4")
  message(FATAL_ERROR "shared FP32 experiment requires ascend910b4")
endif()
if(NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "ASCEND_HOME_PATH required")
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "nonempty CANN build type" FORCE)
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "Ascend compute unit" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
if(OSCAR_SHARED_FP32_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  add_executable(oscar_shared_fp32_producer_cpu probe_producer.cpp producer.cpp)
  add_executable(oscar_shared_fp32_consumer_cpu probe_consumer.cpp producer.cpp consumer.cpp
    ../../kernels/attention_tasks.cpp ../../kernels/attention_cv.cpp)
  target_link_libraries(oscar_shared_fp32_consumer_cpu PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(oscar_shared_fp32_consumer_cpu SYSTEM AFTER PRIVATE
    "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(oscar_shared_fp32_consumer_cpu PRIVATE -O0 -g)
  target_link_libraries(oscar_shared_fp32_producer_cpu PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(oscar_shared_fp32_producer_cpu SYSTEM AFTER PRIVATE
    "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(oscar_shared_fp32_producer_cpu PRIVATE -O0 -g)
else()
  foreach(candidate "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake"
                    "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
    if(EXISTS "${candidate}/ascendc.cmake")
      set(ASCENDC_CMAKE_DIR "${candidate}")
      break()
    endif()
  endforeach()
  if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "CANN ascendc.cmake missing")
  endif()
  include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
  ascendc_library(oscar_shared_fp32_history_experimental SHARED producer.cpp consumer.cpp
    ../../kernels/attention_cv.cpp ../../kernels/attention_cv_striped_cluster4.cpp)
endif()
