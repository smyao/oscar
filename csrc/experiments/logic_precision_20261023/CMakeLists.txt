# Archive #126/#145/#148/#164 and D.4: compile only the isolated CPU harness.
# Link source-bound existing production objects; no kernel or timer rebuild.
cmake_minimum_required(VERSION 3.18)
project(oscar_history_tail LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "match completed CPU harness" FORCE)
if(NOT ASCEND_HOME_PATH OR NOT OSCAR_COMPLETED_BUILD)
  message(FATAL_ERROR "Existing local CANN SDK and completed CPU build required")
endif()
find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
set(needed attention_cv_official_cluster4.cpp.o attention_cv_official_cluster16.cpp.o
           attention_official_tasks.cpp.o merge_lse.cpp.o)
file(GLOB_RECURSE available "${OSCAR_COMPLETED_BUILD}/CMakeFiles/oscar_official_clusters_cpu.dir/*.o")
set(objects)
foreach(name IN LISTS needed)
  set(found)
  foreach(path IN LISTS available)
    get_filename_component(base "${path}" NAME)
    if(base STREQUAL name)
      list(APPEND found "${path}")
    endif()
  endforeach()
  list(LENGTH found count)
  if(NOT count EQUAL 1)
    message(FATAL_ERROR "Exactly one completed production object required: ${name}")
  endif()
  list(APPEND objects ${found})
endforeach()
set_source_files_properties(${objects} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
add_executable(oscar_history_tail_cpu history_only_probe.cpp ${objects})
target_link_libraries(oscar_history_tail_cpu PRIVATE tikicpulib::Ascend910B4)
target_include_directories(oscar_history_tail_cpu SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
target_compile_options(oscar_history_tail_cpu PRIVATE -O0 -g)
