cmake_minimum_required(VERSION 3.18)
project(oscar_request_windows_attention LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "CANN build type" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "requires ASCEND_HOME_PATH and SOC_VERSION=ascend910b4")
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target compute" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
if(OSCAR_REQUEST_WINDOWS_ATTENTION_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  add_executable(oscar_request_window_attention_cpu cpu_probe.cpp decode_bundle.cpp
    ../../kernels/attention_cv_decode_bundle.cpp ../../kernels/attention_tasks.cpp ../../kernels/merge_lse.cpp)
  target_link_libraries(oscar_request_window_attention_cpu PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(oscar_request_window_attention_cpu PRIVATE ../../kernels)
  target_include_directories(oscar_request_window_attention_cpu SYSTEM AFTER PRIVATE
    "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(oscar_request_window_attention_cpu PRIVATE -O0 -g)
  add_executable(oscar_request_window_c16_cpu EXCLUDE_FROM_ALL c16_cpu_probe.cpp window_range_cluster16.cpp
    ../../kernels/attention_cv_window_range_cluster16.cpp ../../kernels/attention_tasks.cpp ../../kernels/merge_lse.cpp)
  target_link_libraries(oscar_request_window_c16_cpu PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(oscar_request_window_c16_cpu PRIVATE ../../kernels)
  target_include_directories(oscar_request_window_c16_cpu SYSTEM AFTER PRIVATE
    "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(oscar_request_window_c16_cpu PRIVATE -O0 -g)
else()
  foreach(candidate "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake"
                    "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
    if(EXISTS "${candidate}/ascendc.cmake")
      set(ASCENDC_CMAKE_DIR "${candidate}")
      break()
    endif()
  endforeach()
  if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "CANN ascendc.cmake missing")
  endif()
  include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
  ascendc_library(oscar_request_window_attention SHARED decode_bundle.cpp ../../kernels/attention_cv_decode_bundle.cpp)
  target_include_directories(oscar_request_window_attention PRIVATE ../../kernels)
endif()
