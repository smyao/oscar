# Independent full-C4 publisher/consumer research, never a deployment target.
cmake_minimum_required(VERSION 3.18)
project(oscar_c4_b_nz LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "research build" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "Requires ascend910b4 and ASCEND_HOME_PATH")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(script generate.py generate_entrypoints.py)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/${script}" --check RESULT_VARIABLE rc)
  if(NOT rc EQUAL 0)
    message(FATAL_ERROR "Research source/entrypoint restoration failed: ${script}")
  endif()
endforeach()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" --check
  --build-binding "${CMAKE_BINARY_DIR}/source_binding.json" RESULT_VARIABLE binding_rc)
if(NOT binding_rc EQUAL 0)
  message(FATAL_ERROR "Research build input binding failed")
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
if(OSCAR_C4_B_NZ_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/publisher_probe.py" --check RESULT_VARIABLE publisher_rc)
  if(NOT publisher_rc EQUAL 0)
    message(FATAL_ERROR "Publisher probe methods differ from final C4 candidate")
  endif()
  add_executable(oscar_c4_b_nz_publisher_cpu publisher_probe.cpp)
  target_link_libraries(oscar_c4_b_nz_publisher_cpu PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(oscar_c4_b_nz_publisher_cpu SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(oscar_c4_b_nz_publisher_cpu PRIVATE -O0 -g)
  add_executable(oscar_c4_b_nz_cpu cv_probe.cpp c4_baseline.cpp c4_candidate.cpp ../../kernels/attention_tasks.cpp)
  target_link_libraries(oscar_c4_b_nz_cpu PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(oscar_c4_b_nz_cpu SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(oscar_c4_b_nz_cpu PRIVATE -O0 -g)
else()
  foreach(path "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake" "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
    if(EXISTS "${path}/ascendc.cmake")
      set(ASCENDC_CMAKE_DIR "${path}")
      break()
    endif()
  endforeach()
  if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "CANN ascendc.cmake missing")
  endif()
  include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
  ascendc_library(oscar_c4_b_nz SHARED c4_baseline.cpp c4_candidate.cpp)
endif()
