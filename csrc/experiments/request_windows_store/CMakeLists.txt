# Archive #21/#126/#145/#154/#155, D.4: isolated experiments; CPU != NPU proof.
cmake_minimum_required(VERSION 3.18)
project(oscar_request_windows_store LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "isolated build type" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4")
  message(FATAL_ERROR "This experiment requires SOC_VERSION=ascend910b4")
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target compute" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
if(OSCAR_REQUEST_WINDOWS_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  add_executable(oscar_request_windows_store_cpu store_cpu.cpp packed_store.cpp window_store.cpp
    ../../kernels/rotate_clip_store_striped.cpp)
  target_link_libraries(oscar_request_windows_store_cpu PRIVATE tikicpulib::Ascend910B4)
  target_compile_options(oscar_request_windows_store_cpu PRIVATE -O0 -g)
else()
  foreach(candidate "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake"
                    "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
    if(EXISTS "${candidate}/ascendc.cmake")
      set(ASCENDC_CMAKE_DIR "${candidate}")
      break()
    endif()
  endforeach()
  if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "CANN ascendc.cmake missing")
  endif()
  include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
  ascendc_library(oscar_request_windows_store SHARED packed_store.cpp window_store.cpp)
endif()
