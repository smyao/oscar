# Archive G7-G13/#126/#145/#148/#164: isolated functional build, never timing.
cmake_minimum_required(VERSION 3.18)
project(oscar_official_math LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "independent functional build" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "Requires local official ascend910b4 SDK")
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
if(OSCAR_OFFICIAL_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  add_executable(oscar_official_decode_cpu probe.cpp ../../kernels/attention_cv_official_decode.cpp
    ../../kernels/store_aligned_int2.cpp ../../kernels/attention_official_tasks.cpp ../../kernels/merge_lse.cpp)
  add_executable(oscar_official_clusters_cpu cluster_probe.cpp ../../kernels/attention_cv_official_cluster4.cpp
    ../../kernels/attention_cv_official_cluster16.cpp ../../kernels/attention_official_tasks.cpp ../../kernels/merge_lse.cpp)
  foreach(target oscar_official_decode_cpu oscar_official_clusters_cpu)
    target_link_libraries(${target} PRIVATE tikicpulib::Ascend910B4)
    target_include_directories(${target} SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
    target_compile_options(${target} PRIVATE -O0 -g)
  endforeach()
  return()
endif()
foreach(path "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake" "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
  if(EXISTS "${path}/ascendc.cmake")
    set(ASCENDC_CMAKE_DIR "${path}")
    break()
  endif()
endforeach()
if(NOT ASCENDC_CMAKE_DIR)
  message(FATAL_ERROR "CANN ascendc.cmake missing")
endif()
include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
ascendc_library(oscar_official_math SHARED ../../kernels/attention_cv_official_decode.cpp
  ../../kernels/attention_cv_official_cluster4.cpp ../../kernels/attention_cv_official_cluster16.cpp)
