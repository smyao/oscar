# Archive G3-G13/#81-85/#126/#145/#154, D.4: independent AIC experiment.
cmake_minimum_required(VERSION 3.18)
project(oscar_decode_mmad_barrier LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "isolated build" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "Requires explicit ascend910b4 and ASCEND_HOME_PATH")
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../../kernels/attention_cv_decode_bundle.cpp" baseline_sha)
if(NOT baseline_sha STREQUAL "d8171f0403cf176087e1c76c64e454e6fb1e74fd79252aab3b59e848cca19401")
  message(FATAL_ERROR "Decode source changed; re-audit baseline configuration")
endif()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/kernel.cpp" kernel_sha)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../../kernels/oscar_common.h" common_sha)
set(marker_json false)
set(cpu_json false)
if(OSCAR_MMAD_CPU_DEBUG)
  set(cpu_json true)
elseif(OSCAR_MMAD_MARKERS)
  set(marker_json true)
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source_binding.json"
  "{\"kernel_sha256\":\"${kernel_sha}\",\"baseline_sha256\":\"${baseline_sha}\",\"common_sha256\":\"${common_sha}\",\"cpu_debug\":${cpu_json},\"markers\":${marker_json}}\n")
if(OSCAR_MMAD_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  add_executable(oscar_decode_mmad_cpu probe.cpp kernel.cpp)
  target_link_libraries(oscar_decode_mmad_cpu PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(oscar_decode_mmad_cpu SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(oscar_decode_mmad_cpu PRIVATE -O0 -g)
else()
  foreach(candidate "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake"
                    "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
    if(EXISTS "${candidate}/ascendc.cmake")
      set(ASCENDC_CMAKE_DIR "${candidate}")
      break()
    endif()
  endforeach()
  if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "CANN ascendc.cmake missing")
  endif()
  include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
  ascendc_library(oscar_decode_mmad_barrier SHARED kernel.cpp)
  if(OSCAR_MMAD_MARKERS)
    ascendc_compile_definitions(oscar_decode_mmad_barrier PRIVATE OSCAR_MMAD_MARKERS=1)
  endif()
endif()
