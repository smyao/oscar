# Archive #126/#145/#148/#164: independent real-SDK finite-check comparison;
# no timing, model, native-source change, or relaxed arithmetic contract.
cmake_minimum_required(VERSION 3.18)
project(oscar_c16_finite_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "ASCEND_HOME_PATH must name the official CANN installation")
endif()
get_filename_component(OSCAR_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(OSCAR_C16_FINITE_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  add_executable(oscar_c16_finite_cpu main.cpp kernel.cpp)
  target_link_libraries(oscar_c16_finite_cpu PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(oscar_c16_finite_cpu PRIVATE "${OSCAR_REPO_ROOT}/csrc/kernels")
  target_include_directories(oscar_c16_finite_cpu SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(oscar_c16_finite_cpu PRIVATE -O0 -g)
else()
  set(ASCEND_COMPUTE_UNIT ascend910b4 CACHE STRING "A2 contract" FORCE)
  set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
  set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
  foreach(candidate "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake"
                    "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake"
                    "${ASCEND_HOME_PATH}/ascendc_devkit/tikcpp/samples/cmake")
    if(EXISTS "${candidate}/ascendc.cmake")
      set(ASCENDC_CMAKE_DIR "${candidate}")
      break()
    endif()
  endforeach()
  if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "official ascendc.cmake is missing")
  endif()
  include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
  ascendc_library(oscar_c16_finite_kernel SHARED kernel.cpp)
  target_include_directories(oscar_c16_finite_kernel PRIVATE "${OSCAR_REPO_ROOT}/csrc/kernels")
endif()
