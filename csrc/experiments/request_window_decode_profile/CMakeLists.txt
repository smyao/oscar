# Archive #132-135/#145/#150 and startup D.4: isolated normal/profile bundle.
# No production target; no native source modifications. Check source identity.
cmake_minimum_required(VERSION 3.18)
project(oscar_request_window_decode_profile LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "isolated diagnostic build" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "Requires ASCEND_HOME_PATH and SOC_VERSION=ascend910b4")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" --check
  RESULT_VARIABLE profile_check_rc)
if(NOT profile_check_rc EQUAL 0)
  message(FATAL_ERROR "Pinned source/generated diagnostic identity check failed")
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
foreach(candidate "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake"
                  "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
  if(EXISTS "${candidate}/ascendc.cmake")
    set(ASCENDC_CMAKE_DIR "${candidate}")
    break()
  endif()
endforeach()
if(NOT ASCENDC_CMAKE_DIR)
  message(FATAL_ERROR "CANN ascendc.cmake missing")
endif()
include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
ascendc_library(oscar_request_window_decode_profile SHARED normal.cpp profile.cpp)
if(OSCAR_CAMODEL_MARKERS)
  ascendc_compile_definitions(oscar_request_window_decode_profile PRIVATE OSCAR_CAMODEL_MARKERS=1)
endif()
