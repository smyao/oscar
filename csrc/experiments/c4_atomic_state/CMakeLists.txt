# Archive G7/G8/#79-85/#126/#145/#154 and D.4: isolated FP32 state legality probe.
# No production registrations, no dtype relaxation, no implicit target run.
cmake_minimum_required(VERSION 3.18)
project(oscar_c4_atomic_state LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "isolated build" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "Requires SOC_VERSION=ascend910b4 and ASCEND_HOME_PATH")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(OSCAR_C4_ATOMIC_READBACK_V2 AND OSCAR_C4_ATOMIC_RECOVERY_V3)
  message(FATAL_ERROR "Choose one independent atomic probe revision")
endif()
if(OSCAR_C4_ATOMIC_RECOVERY_V3)
  set(probe_fixture "fixture_recovery.py")
  set(probe_kernel "kernel_recovery.cpp")
  set(probe_driver "probe_recovery.cpp")
  set(probe_target "oscar_c4_atomic_state_recovery")
elseif(OSCAR_C4_ATOMIC_READBACK_V2)
  set(probe_fixture "fixture_readback.py")
  set(probe_kernel "kernel_readback.cpp")
  set(probe_driver "probe_readback.cpp")
  set(probe_target "oscar_c4_atomic_state_readback")
else()
  set(probe_fixture "fixture.py")
  set(probe_kernel "kernel.cpp")
  set(probe_driver "probe.cpp")
  set(probe_target "oscar_c4_atomic_state")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/${probe_fixture}" --check-source
  RESULT_VARIABLE source_rc OUTPUT_VARIABLE probe_source_json)
if(NOT source_rc EQUAL 0)
  message(FATAL_ERROR "Current C4 static config/source identity check failed")
endif()
if(OSCAR_C4_ATOMIC_RECOVERY_V3)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/atomic_source_binding.json" "${probe_source_json}")
endif()
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "target" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
if(OSCAR_C4_ATOMIC_CPU_DEBUG)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  add_executable(${probe_target}_cpu ${probe_driver} ${probe_kernel})
  target_link_libraries(${probe_target}_cpu PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(${probe_target}_cpu SYSTEM AFTER PRIVATE
    "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(${probe_target}_cpu PRIVATE -O0 -g)
else()
  foreach(candidate "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake"
                    "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
    if(EXISTS "${candidate}/ascendc.cmake")
      set(ASCENDC_CMAKE_DIR "${candidate}")
      break()
    endif()
  endforeach()
  if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "CANN ascendc.cmake missing")
  endif()
  include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
  ascendc_library(${probe_target} SHARED ${probe_kernel})
endif()
