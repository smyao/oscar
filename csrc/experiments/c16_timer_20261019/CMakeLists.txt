# Archive G7-G13/#145/#164: isolated generated C16 only; production untouched.
cmake_minimum_required(VERSION 3.18)
project(oscar_c16_timer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "build" FORCE)
if(NOT SOC_VERSION STREQUAL "ascend910b4" OR NOT ASCEND_HOME_PATH)
  message(FATAL_ERROR "C16 timer requires explicit ascend910b4 CANN")
endif()
set(ASCEND_COMPUTE_UNIT "ascend910b4" CACHE STRING "target" FORCE)
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(ENV{ASCEND_CANN_PACKAGE_PATH} "${ASCEND_HOME_PATH}")
get_filename_component(OSCAR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
get_filename_component(OSCAR_TIMER_GENERATED_DIR "${OSCAR_TIMER_KERNEL_SOURCE}" DIRECTORY)
if(NOT EXISTS "${OSCAR_TIMER_KERNEL_SOURCE}" OR NOT EXISTS "${OSCAR_TIMER_GENERATED_DIR}/bindings.cpp")
  message(FATAL_ERROR "Run generate.py to an external build directory first")
endif()
if(OSCAR_C16_TIMER_CPU)
  find_package(tikicpulib REQUIRED PATHS "${ASCEND_HOME_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
  add_executable(oscar_c16_timer_cpu "${OSCAR_TIMER_CPU_PROBE}" "${OSCAR_TIMER_KERNEL_SOURCE}"
    "${OSCAR_ROOT}/csrc/kernels/attention_tasks.cpp" "${OSCAR_ROOT}/csrc/kernels/merge_lse.cpp")
  target_link_libraries(oscar_c16_timer_cpu PRIVATE tikicpulib::Ascend910B4)
  target_include_directories(oscar_c16_timer_cpu PRIVATE "${OSCAR_ROOT}/csrc/kernels" "${CMAKE_CURRENT_SOURCE_DIR}" "${OSCAR_TIMER_GENERATED_DIR}")
  target_include_directories(oscar_c16_timer_cpu SYSTEM AFTER PRIVATE "${ASCEND_HOME_PATH}/aarch64-linux/asc/include")
  target_compile_options(oscar_c16_timer_cpu PRIVATE -O0 -g)
  return()
endif()
foreach(candidate "${ASCEND_HOME_PATH}/tools/tikcpp/ascendc_kernel_cmake"
    "${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake"
    "${ASCEND_HOME_PATH}/ascendc_devkit/tikcpp/samples/cmake")
  if(EXISTS "${candidate}/ascendc.cmake")
    set(ASCENDC_CMAKE_DIR "${candidate}")
    break()
  endif()
endforeach()
if(NOT ASCENDC_CMAKE_DIR)
  message(FATAL_ERROR "AscendC CMake missing")
endif()
include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
include_directories("${OSCAR_ROOT}/csrc/kernels" "${CMAKE_CURRENT_SOURCE_DIR}" "${OSCAR_TIMER_GENERATED_DIR}")
ascendc_library(oscar_c16_timer_kernels SHARED "${OSCAR_TIMER_KERNEL_SOURCE}")
if(NOT OSCAR_BUILD_SIGNATURE MATCHES "^[0-9a-f]+$")
  message(FATAL_ERROR "Use tools.build_c16_timer to bind the generated binary")
endif()
string(SUBSTRING "${OSCAR_BUILD_SIGNATURE}" 0 16 BINARY_ID)
set_target_properties(oscar_c16_timer_kernels PROPERTIES OUTPUT_NAME "oscar_c16_timer_kernels_${BINARY_ID}")
set(CMAKE_SKIP_RPATH FALSE)
set(CMAKE_SKIP_INSTALL_RPATH FALSE)
set(CMAKE_SKIP_BUILD_RPATH FALSE)
target_link_libraries(oscar_c16_timer_kernels PRIVATE ascendcl runtime profapi unified_dlog mmpa)
target_link_options(oscar_c16_timer_kernels PRIVATE "LINKER:-z,defs")
if(OSCAR_C16_TIMER_KERNEL_ONLY)
  return()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import torch; print(torch.utils.cmake_prefix_path)"
  RESULT_VARIABLE PREFIX_RC OUTPUT_VARIABLE TORCH_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT PREFIX_RC EQUAL 0)
  message(FATAL_ERROR "Selected Python must provide Torch")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${TORCH_PREFIX}")
find_package(Torch REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
add_library(oscar_c16_timer SHARED "${OSCAR_TIMER_GENERATED_DIR}/bindings.cpp")
target_include_directories(oscar_c16_timer PRIVATE "${OSCAR_ROOT}/csrc" ${TORCH_INCLUDE_DIRS}
  "${TORCH_NPU_PATH}/include" "${ASCEND_HOME_PATH}/include")
target_link_directories(oscar_c16_timer PRIVATE "${TORCH_NPU_PATH}/lib" "${ASCEND_HOME_PATH}/lib64")
target_link_libraries(oscar_c16_timer PRIVATE oscar_c16_timer_kernels ${TORCH_LIBRARIES} torch_npu ascendcl Python3::Module)
set_target_properties(oscar_c16_timer PROPERTIES OUTPUT_NAME "oscar_c16_timer_${BINARY_ID}"
  BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN;$ORIGIN/lib;$<TARGET_FILE_DIR:oscar_c16_timer_kernels>;${TORCH_NPU_PATH}/lib;${ASCEND_HOME_PATH}/lib64;${TORCH_INSTALL_PREFIX}/lib")
set_target_properties(oscar_c16_timer_kernels PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "${ASCEND_HOME_PATH}/lib64")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/build_manifest.json" CONTENT
"{\"library\":\"$<TARGET_FILE:oscar_c16_timer>\",\"kernel_library\":\"$<TARGET_FILE:oscar_c16_timer_kernels>\",\"soc\":\"ascend910b4\"}\n")
